-- ========================================================
-- Apartment Maintenance Request Management System Schema
-- ========================================================

CREATE DATABASE IF NOT EXISTS apartment_maintenance_db;
USE apartment_maintenance_db;

-- Table 1: Residents
CREATE TABLE IF NOT EXISTS residents (
    resident_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    flat_no VARCHAR(20) NOT NULL,
    phone VARCHAR(15) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE
);

-- Table 2: Maintenance Staff
CREATE TABLE IF NOT EXISTS maintenance_staff (
    staff_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE
);

-- Table 3: Maintenance Requests
CREATE TABLE IF NOT EXISTS requests (
    request_id INT AUTO_INCREMENT PRIMARY KEY,
    resident_id INT NOT NULL,
    staff_id INT DEFAULT NULL,
    issue_type VARCHAR(50) NOT NULL, -- e.g., 'Plumbing', 'Electrical', 'Cleaning', 'Other'
    request_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(30) DEFAULT 'Pending', -- 'Pending', 'Assigned', 'In Progress', 'Completed'
    description TEXT,
    CONSTRAINT fk_request_resident FOREIGN KEY (resident_id) 
        REFERENCES residents(resident_id) ON DELETE CASCADE,
    CONSTRAINT fk_request_staff FOREIGN KEY (staff_id) 
        REFERENCES maintenance_staff(staff_id) ON DELETE SET NULL
);

-- Table 4: Maintenance Record
CREATE TABLE IF NOT EXISTS maintenance_record (
    record_id INT AUTO_INCREMENT PRIMARY KEY,
    request_id INT NOT NULL UNIQUE,
    completion_date DATETIME NOT NULL,
    repair_cost DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    remarks TEXT,
    CONSTRAINT fk_record_request FOREIGN KEY (request_id) 
        REFERENCES requests(request_id) ON DELETE CASCADE
);
